```csharp:src/40 Nels.Aigc/Nels.Aigc.Domain/Entities/YourEntity.cs
using System;
using System.ComponentModel.DataAnnotations;
using Volo.Abp.Domain.Entities.Auditing;

namespace Nels.Aigc.Entities;

public class YourEntity : AuditedEntity<Guid>, IAggregateRoot<Guid>
{
    public YourEntity() { }
    public YourEntity(Guid id) : base(id) { }

    [Required]
    [MaxLength(100)]
    public virtual string Name { get; set; } = default!;

    [MaxLength(500)]
    public virtual string Description { get; set; } = default!;

    [Required]
    public virtual Guid SpaceId { get; set; } = default!;

    public virtual bool IsEnabled { get; set; } = true;

    // 导航属性
    public virtual List<YourChildEntity> Children { get; set; } = [];

    // 业务方法
    public void AddChild(Guid id, string name)
    {
        if (Children.Exists(x => x.Name == name))
        {
            return;
        }
        Children.Add(new YourChildEntity(id, this.Id, name));
    }

    public void RemoveChild(Guid childId)
    {
        var child = Children.Find(x => x.Id == childId);
        if (child is not null)
        {
            Children.Remove(child);
        }
    }
}

public class YourChildEntity : Entity<Guid>
{
    protected YourChildEntity() { }
    internal YourChildEntity(Guid id, Guid parentId, string name) : base(id)
    {
        ParentId = parentId;
        Name = name;
    }

    public virtual Guid ParentId { get; set; }

    [Required]
    [MaxLength(100)]
    public virtual string Name { get; set; } = default!;
}
```

```csharp:src/40 Nels.Aigc/Nels.Aigc.Application/Services/YourEntityAppService.cs
using Microsoft.AspNetCore.Mvc;
using Nels.Abp.Ddd.Application.Services;
using Nels.Aigc.Dtos;
using Nels.Aigc.Entities;
using Nels.Aigc.Permissions;
using System;
using System.Linq;
using System.Threading.Tasks;
using Volo.Abp.Domain.Repositories;

namespace Nels.Aigc.Services;

[Route(AigcRemoteServiceConsts.yourEntityRoute)]
public class YourEntityAppService : RouteCrudGetAllAppService<YourEntity, YourEntityDto, Guid, YourEntityGetListInputDto, YourEntityCreateDto, YourEntityUpdateDto>, IYourEntityService
{
    public YourEntityAppService(IRepository<YourEntity, Guid> repository) : base(repository)
    {
        CreatePolicyName = AigcPermissions.YourEntity.Create;
        UpdatePolicyName = AigcPermissions.YourEntity.Update;
        DeletePolicyName = AigcPermissions.YourEntity.Delete;
        GetPolicyName = AigcPermissions.YourEntity.Default;
        GetListPolicyName = AigcPermissions.YourEntity.Default;
    }

    [HttpPost]
    [Route("[action]")]
    public async Task SetIsEnabledAsync(Guid id, bool isEnabled)
    {
        var entity = await Repository.GetAsync(id);
        entity.IsEnabled = isEnabled;
        await Repository.UpdateAsync(entity);
    }

    protected override async Task<IQueryable<YourEntity>> CreateFilteredQueryAsync(YourEntityGetListInputDto? input)
    {
        var query = await base.CreateFilteredQueryAsync(input);
        if (input == null) return query;

        return query
            .WhereIf(!string.IsNullOrWhiteSpace(input.Keyword), x => x.Name.Contains(input.Keyword))
            .WhereIf(input.IsEnabled != null, x => x.IsEnabled == input.IsEnabled);
    }

    protected override IQueryable<YourEntity> ApplySorting(IQueryable<YourEntity> query, YourEntityGetListInputDto? input)
    {
        return query.OrderBy(x => x.Name);
    }
}
```

#### 2.3 创建列表页面组件

```typescript:frontend/src/pages/YourEntity/index.tsx
import { getYourEntityList, deleteYourEntity } from '@/services/aigc/yourEntity';
import type { YourEntity, YourEntityFilter } from '@/types/yourEntity';
import { PlusOutlined } from '@ant-design/icons';
import { PageContainer } from '@ant-design/pro-components';
import { Button, Empty, Input, message, Pagination, Skeleton, Switch } from 'antd';
import React, { useCallback, useEffect, useMemo, useState } from 'react';
import { YourEntityCard } from './components/your-entity-card';
import { CreateModal } from './components/create-modal';

const DEFAULT_PAGE_SIZE = 12;

interface SearchState {
  keyword: string;
  isEnabled?: boolean;
  currentPage: number;
  pageSize: number;
}

const YourEntityList: React.FC = () => {
  const [isLoading, setIsLoading] = useState(false);
  const [entities, setEntities] = useState<YourEntity[]>([]);
  const [totalCount, setTotalCount] = useState(0);
  const [isCreateModalOpen, setIsCreateModalOpen] = useState(false);
  const [searchState, setSearchState] = useState<SearchState>({
    keyword: '',
    isEnabled: undefined,
    currentPage: 1,
    pageSize: DEFAULT_PAGE_SIZE,
  });

  const [messageApi] = message.useMessage();

  const loadEntities = useCallback(async () => {
    setIsLoading(true);
    try {
      const filter: YourEntityFilter = {
        keyword: searchState.keyword || undefined,
        isEnabled: searchState.isEnabled,
        skipCount: (searchState.currentPage - 1) * searchState.pageSize,
        maxResultCount: searchState.pageSize,
      };

      const response = await getYourEntityList(filter);
      setEntities(response.items);
      setTotalCount(response.totalCount);
    } catch (error) {
      messageApi.error('加载数据失败');
    } finally {
      setIsLoading(false);
    }
  }, [searchState, messageApi]);

  useEffect(() => {
    loadEntities();
  }, [loadEntities]);

  const handleSearch = (keyword: string) => {
    setSearchState(prev => ({
      ...prev,
      keyword,
      currentPage: 1,
    }));
  };

  const handleDelete = async (id: string) => {
    try {
      await deleteYourEntity(id as UUID);
      messageApi.success('删除成功');
      loadEntities();
    } catch (error) {
      messageApi.error('删除失败');
    }
  };

  return (
    <PageContainer>
      <div className="mb-4 flex justify-between items-center">
        <div className="flex gap-4">
          <Input.Search
            placeholder="搜索名称"
            allowClear
            onSearch={handleSearch}
            style={{ width: 300 }}
          />
          <Switch
            checkedChildren="已启用"
            unCheckedChildren="已禁用"
            onChange={(checked) => setSearchState(prev => ({
              ...prev,
              isEnabled: checked ? true : undefined,
              currentPage: 1,
            }))}
          />
        </div>
        <Button
          type="primary"
          icon={<PlusOutlined />}
          onClick={() => setIsCreateModalOpen(true)}
        >
          新建
        </Button>
      </div>

      {isLoading ? (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
          {Array.from({ length: 8 }).map((_, index) => (
            <Skeleton key={index} active />
          ))}
        </div>
      ) : entities.length === 0 ? (
        <Empty description="暂无数据" />
      ) : (
        <>
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
            {entities.map((entity) => (
              <YourEntityCard
                key={entity.id}
                entity={entity}
                onDelete={handleDelete}
                onChange={loadEntities}
              />
            ))}
          </div>
          <div className="mt-6 flex justify-center">
            <Pagination
              current={searchState.currentPage}
              pageSize={searchState.pageSize}
              total={totalCount}
              showSizeChanger
              showQuickJumper
              showTotal={(total, range) => `第 ${range[0]}-${range[1]} 条/共 ${total} 条`}
              onChange={(page, size) => {
                setSearchState(prev => ({
                  ...prev,
                  currentPage: page,
                  pageSize: size || DEFAULT_PAGE_SIZE,
                }));
              }}
            />
          </div>
        </>
      )}

      <CreateModal
        open={isCreateModalOpen}
        onOpenChange={setIsCreateModalOpen}
        onChange={loadEntities}
      />
    </PageContainer>
  );
};

export default YourEntityList;
```

```typescript:frontend/src/pages/YourEntity/components/create-modal.tsx
import { createYourEntity, updateYourEntity } from '@/services/aigc/yourEntity';
import { YourEntity, YourEntityCreateRequest, YourEntityUpdateRequest } from '@/types/yourEntity';
import { ModalForm, ProFormText, ProFormTextArea, ProFormSwitch } from '@ant-design/pro-components';
import { message, Form } from 'antd';
import React, { useEffect } from 'react';

interface CreateModalProps {
  open: boolean;
  values?: Partial<YourEntity>;
  type?: 'create' | 'edit';
  onChange: () => void;
  onOpenChange: (visible: boolean) => void;
}

export const CreateModal: React.FC<CreateModalProps> = ({ 
  open, 
  type = 'create', 
  values, 
  onChange, 
  onOpenChange 
}) => {
  const [form] = Form.useForm();

  useEffect(() => {
    if (open && values) {
      form.setFieldsValue(values);
    } else if (open) {
      form.resetFields();
    }
  }, [open, values, form]);

  const handleSubmit = async (formData: any) => {
    try {
      if (type === 'create') {
        const createData: YourEntityCreateRequest = {
          name: formData.name,
          description: formData.description || '',
          spaceId: formData.spaceId || '00000000-0000-0000-0000-000000000000', // 默认空间ID
          isEnabled: formData.isEnabled ?? true,
        };
        await createYourEntity(createData);
        message.success('创建成功');
      } else {
        const updateData: YourEntityUpdateRequest = {
          name: formData.name,
          description: formData.description || '',
          isEnabled: formData.isEnabled ?? true,
        };
        await updateYourEntity(values?.id as any, updateData);
        message.success('更新成功');
      }
      
      onChange();
      onOpenChange(false);
      return true;
    } catch (error) {
      message.error(type === 'create' ? '创建失败' : '更新失败');
      return false;
    }
  };

  return (
    <ModalForm
      title={type === 'create' ? '新建实体' : '编辑实体'}
      open={open}
      form={form}
      onOpenChange={onOpenChange}
      onFinish={handleSubmit}
      modalProps={{
        destroyOnClose: true,
      }}
    >
      <ProFormText
        name="name"
        label="名称"
        placeholder="请输入名称"

        }}
      />

        label="启用状态"
```

## 📝 开发规范

### 后端开发规范

1. **实体命名**: 使用PascalCase，继承适当的基类
2. **DTO命名**: 以Dto结尾，分离输入输出DTO
3. **服务命名**: 以AppService结尾，实现对应接口
4. **权限命名**: 使用模块.功能.操作的格式
5. **路由配置**: 使用RESTful风格的路由

### 前端开发规范

1. **组件命名**: 使用PascalCase，文件名使用kebab-case
2. **类型定义**: 与后端DTO保持一致，使用interface
3. **API服务**: 使用camelCase，返回Promise
4. **状态管理**: 使用React Hooks进行状态管理
5. **样式规范**: 使用Tailwind CSS + Ant Design

### 数据库规范

1. **表命名**: 使用PascalCase
2. **字段命名**: 使用PascalCase
3. **外键命名**: 以Id结尾
4. **索引命名**: IX_表名_字段名
