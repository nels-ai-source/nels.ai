### 2. 服务基类选择指南

#### 2.1 只读服务 - RouteReadOnlyAppService

**适用场景**: 只需要查询功能，不需要增删改操作

```csharp:40%20Nels.Aigc%2FNels.Aigc.Application%2FServices%2FYourEntityReadOnlyAppService.cs
[Route(AigcRemoteServiceConsts.yourEntityRoute)]
public class YourEntityReadOnlyAppService : RouteReadOnlyAppService<YourEntity, YourEntityDto, Guid>
{
    public YourEntityReadOnlyAppService(IReadOnlyRepository<YourEntity, Guid> repository) : base(repository)
    {
        GetPolicyName = AigcPermissions.YourEntity.GetList;
        GetListPolicyName = AigcPermissions.YourEntity.GetList;
    }

    // 重写查询过滤逻辑
    protected override async Task<IQueryable<YourEntity>> CreateFilteredQueryAsync(PagedAndSortedResultRequestDto input)
    {
        var query = await base.CreateFilteredQueryAsync(input);
        
        // 添加自定义过滤条件
        if (input is YourEntityGetListInputDto customInput)
        {
            query = query.WhereIf(!string.IsNullOrWhiteSpace(customInput.Keyword), 
                x => x.Name.Contains(customInput.Keyword));
            query = query.WhereIf(customInput.IsEnabled.HasValue, 
                x => x.IsEnabled == customInput.IsEnabled);
        }
        
        return query;
    }
}
```

#### 2.2 完整CRUD服务 - RouteCrudAppService

**适用场景**: 需要完整的增删改查功能

```csharp:40%20Nels.Aigc%2FNels.Aigc.Application%2FServices%2FYourEntityAppService.cs
[Route(AigcRemoteServiceConsts.yourEntityRoute)]
public class YourEntityAppService : RouteCrudAppService<YourEntity, YourEntityDto, Guid, YourEntityGetListInputDto, YourEntityCreateDto, YourEntityUpdateDto>
{
    public YourEntityAppService(IRepository<YourEntity, Guid> repository) : base(repository)
    {
        CreatePolicyName = AigcPermissions.YourEntity.Create;
        UpdatePolicyName = AigcPermissions.YourEntity.Update;
        DeletePolicyName = AigcPermissions.YourEntity.Delete;
        GetPolicyName = AigcPermissions.YourEntity.GetList;
        GetListPolicyName = AigcPermissions.YourEntity.GetList;
    }

    // 重写创建前的处理逻辑
    protected override Task ProcessCreate(YourEntity entity)
    {
        // 设置创建时的默认值
        entity.CreatedBy = CurrentUser.Id;
        entity.TenantId = CurrentTenant.Id;
        return Task.CompletedTask;
    }

    // 重写更新前的处理逻辑
    protected override Task ProcessUpdate(YourEntityUpdateDto input, YourEntity entity)
    {
        // 更新时的业务逻辑
        entity.LastModifiedBy = CurrentUser.Id;
        return Task.CompletedTask;
    }

    // 重写创建验证逻辑
    protected override async Task ValidateCreate(YourEntity entity)
    {
        // 验证名称唯一性
        var existingEntity = await Repository.FirstOrDefaultAsync(x => x.Name == entity.Name);
        if (existingEntity != null)
        {
            throw new UserFriendlyException($"名称 '{entity.Name}' 已存在");
        }
    }

    // 重写更新验证逻辑
    protected override async Task ValidateUpdate(YourEntityUpdateDto input, YourEntity entity)
    {
        // 验证名称唯一性（排除自身）
        var existingEntity = await Repository.FirstOrDefaultAsync(x => x.Name == entity.Name && x.Id != entity.Id);
        if (existingEntity != null)
        {
            throw new UserFriendlyException($"名称 '{entity.Name}' 已存在");
        }
    }

    // 重写删除验证逻辑
    protected override async Task ValidateDelete(Guid id)
    {
        var entity = await Repository.GetAsync(id);
        if (entity.IsSystemDefault)
        {
            throw new UserFriendlyException("系统默认数据不能删除");
        }
    }

    // 重写查询过滤逻辑
    protected override async Task<IQueryable<YourEntity>> CreateFilteredQueryAsync(YourEntityGetListInputDto input)
    {
        var query = await base.CreateFilteredQueryAsync(input);
        
        return query
            .WhereIf(!string.IsNullOrWhiteSpace(input.Keyword), 
                x => x.Name.Contains(input.Keyword) || x.Description.Contains(input.Keyword))
            .WhereIf(input.IsEnabled.HasValue, x => x.IsEnabled == input.IsEnabled)
            .Where(x => x.SpaceId == input.SpaceId); // 按空间过滤
    }
}
```

#### 2.4 复杂业务场景 - 多DTO类型服务

**适用场景**: 查询和列表需要不同的DTO结构

```csharp:40%20Nels.Aigc%2FNels.Aigc.Application%2FServices%2FYourEntityComplexAppService.cs
[Route(AigcRemoteServiceConsts.yourEntityRoute)]
public class YourEntityComplexAppService : RouteCrudGetAllAppService<YourEntity, YourEntityDetailDto, YourEntityListDto, Guid, YourEntityGetListInputDto, YourEntityCreateDto, YourEntityUpdateDto>
{
    private readonly IRepository<RelatedEntity, Guid> _relatedRepository;

    public YourEntityComplexAppService(
        IRepository<YourEntity, Guid> repository,
        IRepository<RelatedEntity, Guid> relatedRepository) : base(repository)
    {
        CreatePolicyName = AigcPermissions.YourEntity.Create;
        UpdatePolicyName = AigcPermissions.YourEntity.Update;
        DeletePolicyName = AigcPermissions.YourEntity.Delete;
        GetPolicyName = AigcPermissions.YourEntity.GetList;
        GetListPolicyName = AigcPermissions.YourEntity.GetList;
        
        _relatedRepository = relatedRepository;
    }

    // 重写详情映射，包含关联数据
    protected override async Task<YourEntityDetailDto> MapToGetOutputDtoAsync(YourEntity entity)
    {
        var dto = await base.MapToGetOutputDtoAsync(entity);
        
        // 加载关联数据
        var relatedEntities = await _relatedRepository.GetListAsync(x => x.YourEntityId == entity.Id);
        dto.RelatedItems = ObjectMapper.Map<List<RelatedEntity>, List<RelatedItemDto>>(relatedEntities);
        
        return dto;
    }

    // 重写列表映射，优化性能
    protected override async Task<List<YourEntityListDto>> MapToGetListOutputDtosAsync(List<YourEntity> entities)
    {
        var dtos = await base.MapToGetListOutputDtosAsync(entities);
        
        // 批量加载关联数据以优化性能
        var entityIds = entities.Select(x => x.Id).ToList();
        var relatedEntities = await _relatedRepository.GetListAsync(x => entityIds.Contains(x.YourEntityId));
        var relatedGroups = relatedEntities.GroupBy(x => x.YourEntityId).ToDictionary(g => g.Key, g => g.Count());
        
        foreach (var dto in dtos)
        {
            dto.RelatedCount = relatedGroups.GetValueOrDefault(dto.Id, 0);
        }
        
        return dtos;
    }

    // 自定义业务方法
    [HttpPost]
    [Route("[action]")]
    public virtual async Task<List<YourEntityListDto>> GetBySpaceAsync(Guid spaceId)
    {
        var entities = await Repository.GetListAsync(x => x.SpaceId == spaceId && x.IsEnabled);
        return await MapToGetListOutputDtosAsync(entities);
    }
}
```

### 6. 最佳实践

1. **选择合适的基类**: 根据业务需求选择最简单够用的基类
2. **权限控制**: 始终配置适当的权限策略
3. **验证逻辑**: 在 `Validate*` 方法中实现业务验证
4. **性能优化**: 在 `CreateFilteredQueryAsync` 中优化查询
5. **错误处理**: 使用 `UserFriendlyException` 提供友好的错误信息
6. **日志记录**: 在关键操作中添加日志记录
7. **事务处理**: 复杂操作使用 `UnitOfWork` 确保数据一致性
