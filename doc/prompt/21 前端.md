## 1. 核心架构模式

### 1.1 通用模块结构
```
src/pages/{ModuleName}/
├── index.tsx                    # 主页面（列表/网格/树形）
├── detail.tsx                   # 详情页面
├── components/
│   ├── {entity}-card.tsx        # 卡片组件
│   ├── create-modal.tsx         # 创建/编辑模态框
│   ├── {entity}-table.tsx       # 表格组件
│   ├── {entity}-tree.tsx        # 树形组件
│   └── filter-bar.tsx           # 筛选栏组件
├── detail/
│   ├── header.tsx               # 详情页头部
│   ├── main-content.tsx         # 主内容区
│   └── sidebar.tsx              # 侧边栏
└── hooks/
    ├── use{Entity}List.ts       # 列表数据管理
    ├── use{Entity}Detail.ts     # 详情数据管理
    └── use{Entity}Filter.ts     # 筛选逻辑管理
```

### 1.2 支持的页面模式

#### 模式一：卡片列表 + 表单
- **适用场景**: Agent管理、模板管理、项目管理
- **特点**: 卡片展示、搜索筛选、创建编辑

#### 模式二：表格列表 + 表单
- **适用场景**: 用户管理、订单管理、日志管理
- **特点**: 表格展示、批量操作、详细筛选

#### 模式三：左树右网格 + 表单
- **适用场景**: 文件管理、组织架构、分类管理
- **特点**: 树形导航、网格展示、层级管理

#### 模式四：左树右表格 + 表单
- **适用场景**: 权限管理、菜单管理、配置管理
- **特点**: 树形导航、表格展示、层级数据

## 4. 通用Hooks模板

### 4.1 列表数据管理Hook
```typescript:src/hooks/use{Entity}List.ts
import { useState, useCallback, useEffect } from 'react';
import { message } from 'antd';
import { get{Entity}List, delete{Entity} } from '@/services/{module}';
import type { {Entity}, {Entity}Filter } from '@/types/{module}';

interface Use{Entity}ListOptions {
  defaultFilter?: Partial<{Entity}Filter>;
  autoLoad?: boolean;
}

export function use{Entity}List(options: Use{Entity}ListOptions = {}) {
  const { defaultFilter = {}, autoLoad = true } = options;
  
  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<{Entity}[]>([]);
  const [total, setTotal] = useState(0);
  const [filter, setFilter] = useState<{Entity}Filter>({
    skipCount: 0,
    maxResultCount: 20,
    ...defaultFilter,
  });

  // 加载数据
  const loadData = useCallback(async (newFilter?: Partial<{Entity}Filter>) => {
    const queryFilter = newFilter ? { ...filter, ...newFilter } : filter;
    
    try {
      setLoading(true);
      const result = await get{Entity}List(queryFilter);
      setData(result.items);
      setTotal(result.totalCount);
      if (newFilter) {
        setFilter(queryFilter);
      }
    } catch (error) {
      message.error('加载数据失败');
      console.error(error);
    } finally {
      setLoading(false);
    }
  }, [filter]);

  // 刷新数据
  const refresh = useCallback(() => {
    loadData();
  }, [loadData]);

  // 搜索
  const search = useCallback((keyword: string) => {
    loadData({ keyword, skipCount: 0 });
  }, [loadData]);

  // 筛选
  const filterData = useCallback((newFilter: Partial<{Entity}Filter>) => {
    loadData({ ...newFilter, skipCount: 0 });
  }, [loadData]);

  // 分页
  const paginate = useCallback((page: number, pageSize: number) => {
    loadData({
      skipCount: (page - 1) * pageSize,
      maxResultCount: pageSize,
    });
  }, [loadData]);

  // 删除
  const deleteItem = useCallback(async (id: string) => {
    try {
      await delete{Entity}(id);
      message.success('删除成功');
      refresh();
    } catch (error) {
      message.error('删除失败');
      console.error(error);
    }
  }, [refresh]);

  useEffect(() => {
    if (autoLoad) {
      loadData();
    }
  }, []);

  return {
    loading,
    data,
    total,
    filter,
    loadData,
    refresh,
    search,
    filterData,
    paginate,
    deleteItem,
  };
}
```

### 4.2 详情数据管理Hook
```typescript:src/hooks/use{Entity}Detail.ts
import { useState, useCallback, useEffect } from 'react';
import { message } from 'antd';
import { get{Entity}, update{Entity} } from '@/services/{module}';
import type { {Entity} } from '@/types/{module}';

export function use{Entity}Detail(id?: string) {
  const [loading, setLoading] = useState(false);
  const [saving, setSaving] = useState(false);
  const [data, setData] = useState<{Entity} | null>(null);

  // 加载详情
  const loadDetail = useCallback(async (entityId: string) => {
    try {
      setLoading(true);
      const result = await get{Entity}(entityId);
      setData(result);
    } catch (error) {
      message.error('加载详情失败');
      console.error(error);
    } finally {
      setLoading(false);
    }
  }, []);

  // 更新数据
  const updateData = useCallback((updates: Partial<{Entity}>) => {
    setData(prev => prev ? { ...prev, ...updates } : null);
  }, []);

  // 保存数据
  const saveData = useCallback(async () => {
    if (!data) return;
    
    try {
      setSaving(true);
      await update{Entity}(data);
      message.success('保存成功');
    } catch (error) {
      message.error('保存失败');
      console.error(error);
    } finally {
      setSaving(false);
    }
  }, [data]);

  useEffect(() => {
    if (id) {
      loadDetail(id);
    }
  }, [id, loadDetail]);

  return {
    loading,
    saving,
    data,
    loadDetail,
    updateData,
    saveData,
  };
}
```

```typescript:src/pages/{Module}/index.tsx
import React from 'react';
import { PageContainer } from '@ant-design/pro-components';
import { Button, Input, Pagination, Empty, Skeleton } from 'antd';
import { PlusOutlined } from '@ant-design/icons';
import { useIntl } from '@umijs/max';

import { use{Entity}List } from '@/hooks/use{Entity}List';

import { FilterBar } from './components/filter-bar';

  const intl = useIntl();

    loading,
    data,
    total,
    filter,
    search,
    filterData,
    paginate,
    refresh,
  } = use{Entity}List();

  const [createModalOpen, setCreateModalOpen] = useState(false);

  return (
    <PageContainer>

            pageSize={filter.maxResultCount}

        open={createModalOpen}

        onSuccess={refresh}
```

```typescript:src/pages/{Module}/table.tsx
          onChange: (page, pageSize) => {
```

### 5.3 左树右网格页面模板
```typescript:src/pages/{Module}/tree-grid.tsx
import React, { useState } from 'react';
import { PageContainer } from '@ant-design/pro-components';
import { Tree, Button, Input, Empty, Skeleton } from 'antd';
import { PlusOutlined, FolderOutlined, FileOutlined } from '@ant-design/icons';
import { useIntl } from '@umijs/max';

import { use{Entity}Tree } from '@/hooks/use{Entity}Tree';
import { use{Entity}List } from '@/hooks/use{Entity}List';
import { {Entity}Card } from './components/{entity}-card';
import { CreateModal } from './components/create-modal';

const {Entity}TreeGrid: React.FC = () => {
  const intl = useIntl();
  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null);
  const [createModalOpen, setCreateModalOpen] = useState(false);
  
  const { 
    treeData, 
    loading: treeLoading, 
    refresh: refreshTree 
  } = use{Entity}Tree();
  
  const {
    data,
    loading: listLoading,
    loadData,
    refresh: refreshList,
  } = use{Entity}List({
    defaultFilter: { parentId: selectedNodeId },
    autoLoad: false,
  });

  // 树节点选择
  const handleTreeSelect = (selectedKeys: React.Key[]) => {
    const nodeId = selectedKeys[0] as string;
    setSelectedNodeId(nodeId);
    loadData({ parentId: nodeId, skipCount: 0 });
  };

  // 渲染树节点图标
  const renderTreeIcon = (node: any) => {
    return node.children?.length > 0 ? <FolderOutlined /> : <FileOutlined />;
  };

  return (
    <PageContainer>
      <div className="flex h-full">
        {/* 左侧树形导航 */}
        <div className="w-80 bg-white border-r border-gray-200 p-4">
          <div className="flex items-center justify-between mb-4">
            <h3 className="text-lg font-medium">
              {intl.formatMessage({ id: 'tree.title' })}
            </h3>
            <Button
              icon={<PlusOutlined />}
              size="small"
              onClick={() => setCreateModalOpen(true)}
            />
          </div>
          
          {treeLoading ? (
            <Skeleton active />
          ) : (
            <Tree
              treeData={treeData}
              onSelect={handleTreeSelect}
              selectedKeys={selectedNodeId ? [selectedNodeId] : []}
              icon={renderTreeIcon}
              showIcon
            />
          )}
        </div>

        {/* 右侧网格内容 */}
        <div className="flex-1 p-4">
          <div className="flex items-center justify-between mb-4">
            <Input.Search
              placeholder={intl.formatMessage({ id: 'search.placeholder' })}
              style={{ width: 200 }}
              allowClear
              onSearch={(keyword) => loadData({ keyword, skipCount: 0 })}
            />
            <Button
              icon={<PlusOutlined />}
              type="primary"
              onClick={() => setCreateModalOpen(true)}
              disabled={!selectedNodeId}
            >
              {intl.formatMessage({ id: 'actions.create' })}
            </Button>
          </div>

          {listLoading ? (
            <Skeleton active />
          ) : data.length === 0 ? (
            <Empty description={
              selectedNodeId 
                ? intl.formatMessage({ id: 'empty.noData' })
                : intl.formatMessage({ id: 'empty.selectNode' })
            } />
          ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
              {data.map((item) => (
                <{Entity}Card 
                  key={item.id} 
                  data={item} 
                  onUpdate={refreshList} 
                />
              ))}
            </div>
          )}
        </div>
      </div>

      <CreateModal
        open={createModalOpen}
        onOpenChange={setCreateModalOpen}
        defaultValues={{ parentId: selectedNodeId }}
        onSuccess={() => {
          refreshTree();
          refreshList();
        }}
      />
    </PageContainer>
  );
};

export default {Entity}TreeGrid;
```

## 6. 通用组件模板

### 6.1 筛选栏组件
```typescript:src/components/FilterBar.tsx
import React from 'react';
import { Select, DatePicker, Button } from 'antd';
import { useIntl } from '@umijs/max';

interface FilterBarProps {
  onFilter: (filter: any) => void;
  filters?: {
    type?: boolean;
    status?: boolean;
    dateRange?: boolean;
    // 可扩展其他筛选项
  };
}

export const FilterBar: React.FC<FilterBarProps> = ({ 
  onFilter, 
  filters = {} 
}) => {
  const intl = useIntl();
  const [filterValues, setFilterValues] = useState<any>({});

  const handleFilterChange = (key: string, value: any) => {
    const newValues = { ...filterValues, [key]: value };
    setFilterValues(newValues);
    onFilter(newValues);
  };

  const handleReset = () => {
    setFilterValues({});
    onFilter({});
  };

  return (
    <div className="flex items-center gap-2">
      {filters.type && (
        <Select
          placeholder={intl.formatMessage({ id: 'filter.type' })}
          style={{ width: 120 }}
          allowClear
          value={filterValues.type}
          onChange={(value) => handleFilterChange('type', value)}
        >
          {/* 根据实际枚举渲染选项 */}
        </Select>
      )}
      
      {filters.status && (
        <Select
          placeholder={intl.formatMessage({ id: 'filter.status' })}
          style={{ width: 120 }}
          allowClear
          value={filterValues.status}
          onChange={(value) => handleFilterChange('status', value)}
        >
          {/* 根据实际枚举渲染选项 */}
        </Select>
      )}
      
      {filters.dateRange && (
        <DatePicker.RangePicker
          value={filterValues.dateRange}
          onChange={(value) => handleFilterChange('dateRange', value)}
        />
      )}
      
      <Button onClick={handleReset}>
        {intl.formatMessage({ id: 'actions.reset' })}
      </Button>
    </div>
  );
};
```

## 7. 最佳实践总结

### 7.1 开发流程
1. **定义类型**: 先定义实体类型和接口
2. **创建服务**: 实现API服务层
3. **开发Hooks**: 封装数据管理逻辑
4. **构建组件**: 按模式构建页面组件
5. **添加国际化**: 配置多语言支持
6. **样式优化**: 应用统一样式规范

### 7.2 代码复用策略
1. **模板化开发**: 使用通用模板快速生成代码
2. **组件抽象**: 提取可复用的通用组件
3. **Hook复用**: 封装通用的数据管理逻辑
4. **类型复用**: 定义通用的基础类型

### 7.3 性能优化
1. **懒加载**: 大型页面使用React.lazy
2. **虚拟滚动**: 大数据量列表使用虚拟滚动
3. **缓存策略**: 合理使用React Query或SWR
4. **代码分割**: 按路由和功能分割代码
