# Nels.AI 前端开发指引 - 表格列表模式

## 1. 适用场景
- **用户管理**: 展示用户列表和详细信息
- **订单管理**: 展示订单数据和状态
- **日志管理**: 展示系统日志和操作记录
- **数据管理**: 展示结构化数据

## 2. 页面特点
- **表格展示**: 以表格形式展示结构化数据
- **批量操作**: 支持多选和批量处理
- **高级筛选**: 支持多条件组合筛选
- **排序分页**: 支持列排序和分页

## 4. 表格专用Hooks

### 4.1 表格数据管理Hook
```typescript:src/hooks/use{Entity}Table.ts
import { useState, useCallback } from 'react';
import { message } from 'antd';
import { get{Entity}List, delete{Entity}, delete{Entity}Batch } from '@/services/{module}';
import type { {Entity}, {Entity}Filter } from '@/types/{module}';

export function use{Entity}Table() {
  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<{Entity}[]>([]);
  const [total, setTotal] = useState(0);

  // 加载表格数据
  const loadTableData = useCallback(async (params: any) => {
    try {
      setLoading(true);
      const filter: {Entity}Filter = {
        keyword: params.name, // 搜索关键词
        type: params.type,
        status: params.status,
        skipCount: params.skipCount || 0,
        maxResultCount: params.maxResultCount || 20,
      };

      const result = await get{Entity}List(filter);
      setData(result.items);
      setTotal(result.totalCount);
      
      return {
        data: result.items,
        success: true,
        total: result.totalCount,
      };
    } catch (error) {
      message.error('加载数据失败');
      console.error(error);
      return {
        data: [],
        success: false,
        total: 0,
      };
    } finally {
      setLoading(false);
    }
  }, []);

  // 删除单个项目
  const deleteItem = useCallback(async (id: string) => {
    try {
      await delete{Entity}(id);
      message.success('删除成功');
      return true;
    } catch (error) {
      message.error('删除失败');
      console.error(error);
      return false;
    }
  }, []);

  // 批量删除
  const batchDelete = useCallback(async (ids: string[]) => {
    try {
      await delete{Entity}Batch(ids);
      message.success(`成功删除 ${ids.length} 个项目`);
      return true;
    } catch (error) {
      message.error('批量删除失败');
      console.error(error);
      return false;
    }
  }, []);

  return {
    loading,
    data,
    total,
    loadTableData,
    deleteItem,
    batchDelete,
  };
}
```

import { useState, useCallback } from 'react';

export function useTableSelection<T = any>() {
  const [selectedRowKeys, setSelectedRowKeys] = useState<React.Key[]>([]);
  const [selectedRows, setSelectedRows] = useState<T[]>([]);

  const rowSelection = {
    selectedRowKeys,
    onChange: (newSelectedRowKeys: React.Key[], newSelectedRows: T[]) => {
      setSelectedRowKeys(newSelectedRowKeys);
      setSelectedRows(newSelectedRows);
    },
    onSelect: (record: T, selected: boolean, selectedRows: T[]) => {
      // 单选处理逻辑
    },
    onSelectAll: (selected: boolean, selectedRows: T[], changeRows: T[]) => {
      // 全选处理逻辑
    },
  };

  const clearSelection = useCallback(() => {
    setSelectedRowKeys([]);
    setSelectedRows([]);
  }, []);

  const selectAll = useCallback((allRows: T[], keyField: string = 'id') => {
    const allKeys = allRows.map(row => (row as any)[keyField]);
    setSelectedRowKeys(allKeys);
    setSelectedRows(allRows);
  }, []);

  return {
    selectedRowKeys,
    selectedRows,
    rowSelection,
    clearSelection,
    selectAll,
    hasSelected: selectedRowKeys.length > 0,
    selectedCount: selectedRowKeys.length,
  };
}

### 7.1 Excel导入组件
```typescript:src/components/ExcelImport.tsx
import React, { useState } from 'react';
import { Upload, Button, Modal, Table, message, Progress } from 'antd';
import { InboxOutlined, DownloadOutlined } from '@ant-design/icons';
import * as XLSX from 'xlsx';

interface ExcelImportProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  onImport: (data: any[]) => Promise<void>;
  templateUrl?: string;
  columns: any[];
}

export const ExcelImport: React.FC<ExcelImportProps> = ({
  open,
  onOpenChange,
  onImport,
  templateUrl,
  columns,
}) => {
  const [importData, setImportData] = useState<any[]>([]);
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);

  const handleFileUpload = (file: File) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target?.result as ArrayBuffer);
        const workbook = XLSX.read(data, { type: 'array' });
        const sheetName = workbook.SheetNames[0];
        const worksheet = workbook.Sheets[sheetName];
        const jsonData = XLSX.utils.sheet_to_json(worksheet);
        
        setImportData(jsonData);
      } catch (error) {
        message.error('文件解析失败');
      }
    };
    reader.readAsArrayBuffer(file);
    return false;
  };

  const handleImport = async () => {
    if (importData.length === 0) {
      message.warning('请先上传文件');
      return;
    }

    try {
      setUploading(true);
      setProgress(0);
      
      // 模拟进度
      const timer = setInterval(() => {
        setProgress(prev => {
          if (prev >= 90) {
            clearInterval(timer);
            return prev;
          }
          return prev + 10;
        });
      }, 200);

      await onImport(importData);
      
      clearInterval(timer);
      setProgress(100);
      message.success('导入成功');
      onOpenChange(false);
      setImportData([]);
    } catch (error) {
      message.error('导入失败');
    } finally {
      setUploading(false);
      setProgress(0);
    }
  };

  return (
    <Modal
      title="数据导入"
      open={open}
      onCancel={() => onOpenChange(false)}
      width={800}
      footer={[
        <Button key="cancel" onClick={() => onOpenChange(false)}>
          取消
        </Button>,
        <Button
          key="import"
          type="primary"
          loading={uploading}
          onClick={handleImport}
          disabled={importData.length === 0}
        >
          导入数据
        </Button>,
      ]}
    >
      <div className="space-y-4">
        {templateUrl && (
          <div>
            <Button
              icon={<DownloadOutlined />}
              onClick={() => window.open(templateUrl)}
            >
              下载导入模板
            </Button>
          </div>
        )}
        
        <Upload.Dragger
          accept=".xlsx,.xls"
          beforeUpload={handleFileUpload}
          showUploadList={false}
        >
          <p className="ant-upload-drag-icon">
            <InboxOutlined />
          </p>
          <p className="ant-upload-text">点击或拖拽文件到此区域上传</p>
          <p className="ant-upload-hint">支持 .xlsx 和 .xls 格式</p>
        </Upload.Dragger>
        
        {uploading && (
          <Progress percent={progress} status="active" />
        )}
        
        {importData.length > 0 && (
          <div>
            <h4>预览数据 (共 {importData.length} 条)</h4>
            <Table
              dataSource={importData.slice(0, 10)}
              columns={columns}
              pagination={false}
              size="small"
              scroll={{ x: true }}
            />
            {importData.length > 10 && (
              <p className="text-gray-500 mt-2">
                仅显示前10条数据，实际将导入 {importData.length} 条
              </p>
            )}
          </div>
        )}
      </div>
    </Modal>
  );
};
```
