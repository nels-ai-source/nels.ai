# Nels.AI 前端开发指引 - 左树右网格模式

## 1. 适用场景
- **分类管理**: 左侧分类树，右侧分类下的内容网格
- **组织架构**: 左侧部门树，右侧部门成员网格
- **文件管理**: 左侧目录树，右侧文件网格
- **权限管理**: 左侧菜单树，右侧权限配置网格

## 2. 页面特点
- **树形导航**: 左侧树形结构导航
- **网格展示**: 右侧网格形式展示数据
- **联动交互**: 左右两侧数据联动
- **层级管理**: 支持多级分类管理

### 3.1 基础左树右网格页面
```typescript:src/pages/{Module}/tree-grid.tsx
import React, { useState, useEffect } from 'react';
import { PageContainer } from '@ant-design/pro-components';
import { Row, Col, Card, Tree, Empty, Spin } from 'antd';
import { useIntl } from '@umijs/max';
import type { DataNode } from 'antd/es/tree';

import { use{Entity}Tree } from '@/hooks/use{Entity}Tree';
import { use{Entity}Grid } from '@/hooks/use{Entity}Grid';
import { TreeToolbar } from './components/tree-toolbar';
import { GridView } from './components/grid-view';
import { CreateModal } from './components/create-modal';
import type { {Entity}, {Entity}TreeNode } from '@/types/{module}';

const {Entity}TreeGrid: React.FC = () => {
  const intl = useIntl();
  const [selectedTreeNode, setSelectedTreeNode] = useState<string | null>(null);
  const [createModalOpen, setCreateModalOpen] = useState(false);
  const [editingItem, setEditingItem] = useState<{Entity} | null>(null);

  const {
    treeData,
    loading: treeLoading,
    loadTree,
    addTreeNode,
    updateTreeNode,
    deleteTreeNode,
  } = use{Entity}Tree();

  const {
    gridData,
    loading: gridLoading,
    total,
    loadGridData,
    deleteGridItem,
  } = use{Entity}Grid();

  // 初始化加载树数据
  useEffect(() => {
    loadTree();
  }, [loadTree]);

  // 树节点选择
  const handleTreeSelect = (selectedKeys: React.Key[]) => {
    const nodeId = selectedKeys[0] as string;
    setSelectedTreeNode(nodeId);
    if (nodeId) {
      loadGridData({ categoryId: nodeId });
    }
  };

  // 树数据转换
  const convertToTreeData = (nodes: {Entity}TreeNode[]): DataNode[] => {
    return nodes.map(node => ({
      key: node.id,
      title: node.name,
      children: node.children ? convertToTreeData(node.children) : undefined,
      isLeaf: !node.children || node.children.length === 0,
    }));
  };

  return (
    <PageContainer>
      <Row gutter={16} style={{ height: 'calc(100vh - 200px)' }}>
        {/* 左侧树形导航 */}
        <Col span={6}>
          <Card
            title={intl.formatMessage({ id: '{module}.tree.title' })}
            size="small"
            className="h-full"
            extra={
              <TreeToolbar
                onAdd={() => setCreateModalOpen(true)}
                onRefresh={() => loadTree()}
                selectedNode={selectedTreeNode}
                onEdit={(node) => setEditingItem(node)}
                onDelete={(nodeId) => deleteTreeNode(nodeId)}
              />
            }
          >
            <div className="h-full overflow-auto">
              <Spin spinning={treeLoading}>
                {treeData.length > 0 ? (
                  <Tree
                    treeData={convertToTreeData(treeData)}
                    onSelect={handleTreeSelect}
                    selectedKeys={selectedTreeNode ? [selectedTreeNode] : []}
                    showLine
                    showIcon={false}
                    blockNode
                  />
                ) : (
                  <Empty
                    image={Empty.PRESENTED_IMAGE_SIMPLE}
                    description={intl.formatMessage({ id: 'tree.empty' })}
                  />
                )}
              </Spin>
            </div>
          </Card>
        </Col>

        {/* 右侧网格内容 */}
        <Col span={18}>
          <Card
            title={
              selectedTreeNode
                ? intl.formatMessage({ id: '{module}.grid.title' })
                : intl.formatMessage({ id: '{module}.grid.selectNode' })
            }
            size="small"
            className="h-full"
          >
            <div className="h-full overflow-auto">
              {selectedTreeNode ? (
                <GridView
                  data={gridData}
                  loading={gridLoading}
                  total={total}
                  onEdit={setEditingItem}
                  onDelete={deleteGridItem}
                  onAdd={() => setCreateModalOpen(true)}
                  categoryId={selectedTreeNode}
                />
              ) : (
                <div className="flex items-center justify-center h-full">
                  <Empty
                    description={intl.formatMessage({ id: 'grid.selectCategory' })}
                  />
                </div>
              )}
            </div>
          </Card>
        </Col>
      </Row>

      <CreateModal
        open={createModalOpen || !!editingItem}
        values={editingItem}
        type={editingItem ? 'edit' : 'create'}
        categoryId={selectedTreeNode}
        onOpenChange={(open) => {
          if (!open) {
            setCreateModalOpen(false);
            setEditingItem(null);
          }
        }}
        onSuccess={() => {
          if (selectedTreeNode) {
            loadGridData({ categoryId: selectedTreeNode });
          }
          loadTree();
          setCreateModalOpen(false);
          setEditingItem(null);
        }}
      />
    </PageContainer>
  );
};

export default {Entity}TreeGrid;
```

### 3.2 高级左树右网格页面（带搜索和筛选）

```typescript:src/pages/{Module}/advanced-tree-grid.tsx
import React, { useState, useEffect, useMemo } from 'react';
import { PageContainer } from '@ant-design/pro-components';
import { Row, Col, Card, Tree, Input, Button, Space, Dropdown } from 'antd';
import { SearchOutlined, FilterOutlined, ReloadOutlined } from '@ant-design/icons';
import { useIntl } from '@umijs/max';

const {Entity}AdvancedTreeGrid: React.FC = () => {
  const intl = useIntl();
  const [treeSearchValue, setTreeSearchValue] = useState('');
  const [expandedKeys, setExpandedKeys] = useState<React.Key[]>([]);
  const [autoExpandParent, setAutoExpandParent] = useState(true);

  // 树搜索功能
  const filteredTreeData = useMemo(() => {
    if (!treeSearchValue) return treeData;
    
    const filterTree = (nodes: {Entity}TreeNode[]): {Entity}TreeNode[] => {
      return nodes.filter(node => {
        const match = node.name.toLowerCase().includes(treeSearchValue.toLowerCase());
        const hasMatchingChildren = node.children && filterTree(node.children).length > 0;
        return match || hasMatchingChildren;
      }).map(node => ({
        ...node,
        children: node.children ? filterTree(node.children) : undefined,
      }));
    };
    
    return filterTree(treeData);
  }, [treeData, treeSearchValue]);

  // 搜索时自动展开
  useEffect(() => {
    if (treeSearchValue) {
      const getAllKeys = (nodes: {Entity}TreeNode[]): string[] => {
        let keys: string[] = [];
        nodes.forEach(node => {
          keys.push(node.id);
          if (node.children) {
            keys = keys.concat(getAllKeys(node.children));
          }
        });
        return keys;
      };
      setExpandedKeys(getAllKeys(filteredTreeData));
      setAutoExpandParent(true);
    }
  }, [treeSearchValue, filteredTreeData]);

  const handleTreeExpand = (expandedKeys: React.Key[]) => {
    setExpandedKeys(expandedKeys);
    setAutoExpandParent(false);
  };

  // 筛选菜单
  const filterMenuItems = [
    {
      key: 'all',
      label: '全部',
    },
    {
      key: 'active',
      label: '活跃',
    },
    {
      key: 'inactive',
      label: '非活跃',
    },
  ];

  return (
    <PageContainer>
      <Row gutter={16} style={{ height: 'calc(100vh - 200px)' }}>
        {/* 左侧树形导航 */}
        <Col span={6}>
          <Card
            title={intl.formatMessage({ id: '{module}.tree.title' })}
            size="small"
            className="h-full"
            extra={
              <Space>
                <Button
                  size="small"
                  icon={<ReloadOutlined />}
                  onClick={() => loadTree()}
                />
                <Dropdown menu={{ items: filterMenuItems }}>
                  <Button size="small" icon={<FilterOutlined />} />
                </Dropdown>
              </Space>
            }
          >
            {/* 树搜索 */}
            <div className="mb-3">
              <Input
                placeholder={intl.formatMessage({ id: 'tree.search.placeholder' })}
                prefix={<SearchOutlined />}
                value={treeSearchValue}
                onChange={(e) => setTreeSearchValue(e.target.value)}
                allowClear
              />
            </div>

            {/* 树组件 */}
            <div className="h-full overflow-auto">
              <Tree
                treeData={convertToTreeData(filteredTreeData)}
                onSelect={handleTreeSelect}
                onExpand={handleTreeExpand}
                expandedKeys={expandedKeys}
                autoExpandParent={autoExpandParent}
                selectedKeys={selectedTreeNode ? [selectedTreeNode] : []}
                showLine
                showIcon={false}
                blockNode
                filterTreeNode={(node) => {
                  return node.title?.toString().toLowerCase().includes(treeSearchValue.toLowerCase()) || false;
                }}
              />
            </div>
          </Card>
        </Col>

        {/* 右侧网格内容 */}
        <Col span={18}>
          <Card
            title={
              selectedTreeNode
                ? intl.formatMessage({ id: '{module}.grid.title' })
                : intl.formatMessage({ id: '{module}.grid.selectNode' })
            }
            size="small"
            className="h-full"
          >
            <GridView
              data={gridData}
              loading={gridLoading}
              total={total}
              onEdit={setEditingItem}
              onDelete={deleteGridItem}
              onAdd={() => setCreateModalOpen(true)}
              categoryId={selectedTreeNode}
              showSearch={true}
              showFilter={true}
            />
          </Card>
        </Col>
      </Row>
    </PageContainer>
  );
};

export default {Entity}AdvancedTreeGrid;
```

## 4. 树形导航Hooks

### 4.1 树数据管理Hook
```typescript:src/hooks/use{Entity}Tree.ts
import { useState, useCallback } from 'react';
import { message } from 'antd';
import { 
  get{Entity}Tree, 
  create{Entity}TreeNode, 
  update{Entity}TreeNode, 
  delete{Entity}TreeNode 
} from '@/services/{module}';
import type { {Entity}TreeNode } from '@/types/{module}';

export function use{Entity}Tree() {
  const [treeData, setTreeData] = useState<{Entity}TreeNode[]>([]);
  const [loading, setLoading] = useState(false);

  // 加载树数据
  const loadTree = useCallback(async () => {
    try {
      setLoading(true);
      const result = await get{Entity}Tree();
      setTreeData(result);
    } catch (error) {
      message.error('加载树数据失败');
      console.error(error);
    } finally {
      setLoading(false);
    }
  }, []);

  // 添加树节点
  const addTreeNode = useCallback(async (nodeData: Partial<{Entity}TreeNode>) => {
    try {
      const newNode = await create{Entity}TreeNode(nodeData);
      await loadTree(); // 重新加载树数据
      message.success('添加成功');
      return newNode;
    } catch (error) {
      message.error('添加失败');
      console.error(error);
      throw error;
    }
  }, [loadTree]);

  // 更新树节点
  const updateTreeNode = useCallback(async (id: string, nodeData: Partial<{Entity}TreeNode>) => {
    try {
      const updatedNode = await update{Entity}TreeNode(id, nodeData);
      await loadTree(); // 重新加载树数据
      message.success('更新成功');
      return updatedNode;
    } catch (error) {
      message.error('更新失败');
      console.error(error);
      throw error;
    }
  }, [loadTree]);

  // 删除树节点
  const deleteTreeNode = useCallback(async (id: string) => {
    try {
      await delete{Entity}TreeNode(id);
      await loadTree(); // 重新加载树数据
      message.success('删除成功');
    } catch (error) {
      message.error('删除失败');
      console.error(error);
      throw error;
    }
  }, [loadTree]);

  // 查找节点路径
  const findNodePath = useCallback((nodeId: string, nodes: {Entity}TreeNode[] = treeData): string[] => {
    for (const node of nodes) {
      if (node.id === nodeId) {
        return [node.id];
      }
      if (node.children) {
        const path = findNodePath(nodeId, node.children);
        if (path.length > 0) {
          return [node.id, ...path];
        }
      }
    }
    return [];
  }, [treeData]);

  // 获取所有叶子节点
  const getLeafNodes = useCallback((nodes: {Entity}TreeNode[] = treeData): {Entity}TreeNode[] => {
    let leafNodes: {Entity}TreeNode[] = [];
    nodes.forEach(node => {
      if (!node.children || node.children.length === 0) {
        leafNodes.push(node);
      } else {
        leafNodes = leafNodes.concat(getLeafNodes(node.children));
      }
    });
    return leafNodes;
  }, [treeData]);

  return {
    treeData,
    loading,
    loadTree,
    addTreeNode,
    updateTreeNode,
    deleteTreeNode,
    findNodePath,
    getLeafNodes,
  };
}
```

### 4.2 网格数据管理Hook
```typescript:src/hooks/use{Entity}Grid.ts
import { useState, useCallback } from 'react';
import { message } from 'antd';
import { get{Entity}List, delete{Entity} } from '@/services/{module}';
import type { {Entity}, {Entity}Filter } from '@/types/{module}';

export function use{Entity}Grid() {
  const [gridData, setGridData] = useState<{Entity}[]>([]);
  const [loading, setLoading] = useState(false);
  const [total, setTotal] = useState(0);
  const [currentPage, setCurrentPage] = useState(1);
  const [pageSize, setPageSize] = useState(20);

  // 加载网格数据
  const loadGridData = useCallback(async (filter: Partial<{Entity}Filter> = {}) => {
    try {
      setLoading(true);
      const queryFilter: {Entity}Filter = {
        ...filter,
        skipCount: (currentPage - 1) * pageSize,
        maxResultCount: pageSize,
      };

      const result = await get{Entity}List(queryFilter);
      setGridData(result.items);
      setTotal(result.totalCount);
    } catch (error) {
      message.error('加载数据失败');
      console.error(error);
    } finally {
      setLoading(false);
    }
  }, [currentPage, pageSize]);

  // 删除网格项目
  const deleteGridItem = useCallback(async (id: string) => {
    try {
      await delete{Entity}(id);
      message.success('删除成功');
      // 重新加载当前页数据
      await loadGridData();
    } catch (error) {
      message.error('删除失败');
      console.error(error);
    }
  }, [loadGridData]);

  // 分页处理
  const handlePageChange = useCallback((page: number, size?: number) => {
    setCurrentPage(page);
    if (size && size !== pageSize) {
      setPageSize(size);
    }
  }, [pageSize]);

  return {
    gridData,
    loading,
    total,
    currentPage,
    pageSize,
    loadGridData,
    deleteGridItem,
    handlePageChange,
  };
}
```

## 6. 网格视图组件

```typescript:src/pages/{Module}/components/grid-view.tsx
import React, { useState } from 'react';

import type { {Entity} } from '@/types/{module}';

interface GridViewProps {
  data: {Entity}[];
  loading: boolean;
  total: number;
  categoryId?: string | null;
  onEdit: (item: {Entity}) => void;
  onDelete: (id: string) => void;
  onAdd: () => void;
  showSearch?: boolean;
  showFilter?: boolean;
}

export const GridView: React.FC<GridViewProps> = ({
  data,
  loading,
  total,
  categoryId,
  onEdit,
  onDelete,
  onAdd,
  showSearch = false,
  showFilter = false,
}) => {
  const intl = useIntl();
  const [searchValue, setSearchValue] = useState('');
  const [filterValue, setFilterValue] = useState<string | undefined>();
  const [currentPage, setCurrentPage] = useState(1);

    const matchSearch = !searchValue || 
      item.name.toLowerCase().includes(searchValue.toLowerCase());
    const matchFilter = !filterValue || item.type === filterValue;
    return matchSearch && matchFilter;
  });

    (currentPage - 1) * pageSize,
    currentPage * pageSize
  );

            size="small"
            icon={<EditOutlined />}

            {intl.formatMessage({ id: 'actions.delete' })}
          </Button>,
        ]}
      >
        <Card.Meta
          title={
            <div className="truncate" title={item.name}>
              {item.name}
            </div>
          }
          description={
            <div className="text-gray-500 text-sm">
              <div className="truncate" title={item.description}>
                {item.description}
              </div>
              <div className="mt-1">
                类型: {item.type}
              </div>
            </div>
          }
        />
      </Card>
    </Col>
  );

  if (!categoryId) {
    return (
      <div className="flex items-center justify-center h-full">
        <Empty description={intl.formatMessage({ id: 'grid.selectCategory' })} />
      </div>
    );
  }

  return (
    <div className="h-full flex flex-col">
      {/* 工具栏 */}

        <Space>
          <Button
            type="primary"
            icon={<PlusOutlined />}
            onClick={onAdd}
          >
            {intl.formatMessage({ id: 'actions.add' })}
          </Button>
          
          {showSearch && (
            <Input
              placeholder={intl.formatMessage({ id: 'search.placeholder' })}
              prefix={<SearchOutlined />}
              value={searchValue}
              onChange={(e) => setSearchValue(e.target.value)}
              style={{ width: 200 }}
              allowClear
            />
          )}
          
          {showFilter && (
            <Select
              placeholder={intl.formatMessage({ id: 'filter.type' })}
              value={filterValue}
              onChange={setFilterValue}
              style={{ width: 120 }}
              allowClear
              options={[
                { label: '类型1', value: 'type1' },
                { label: '类型2', value: 'type2' },
                { label: '类型3', value: 'type3' },
              ]}
            />
          )}
        </Space>
        
        <div className="text-gray-500">
          共 {total} 项
        </div>
      </div>

      {/* 网格内容 */}
      <div className="flex-1 overflow-auto">
        {paginatedData.length > 0 ? (
          <Row gutter={[16, 16]}>
            {paginatedData.map(renderGridItem)}
          </Row>
        ) : (
          <Empty description={intl.formatMessage({ id: 'grid.empty' })} />
        )}
      </div>

      {/* 分页 */}
      {total > pageSize && (
        <div className="mt-4 text-center">
          <Pagination
            current={currentPage}
            pageSize={pageSize}
            total={filteredData.length}
            onChange={(page, size) => {
              setCurrentPage(page);
              if (size !== pageSize) {
                setPageSize(size);
                setCurrentPage(1);
              }
            }}
            showSizeChanger
            showQuickJumper
            showTotal={(total, range) =>
              `第 ${range[0]}-${range[1]} 条/总共 ${total} 条`
            }
          />
        </div>
      )}
    </div>
  );
```

### 6.2 高级网格视图（带拖拽排序）
```typescript:src/pages/{Module}/components/draggable-grid-view.tsx

  onDelete: (id: string) => void;
}

const DraggableGridItem: React.FC<DraggableGridItemProps> = ({
  item,
  index,
  onMove,
  onEdit,
  onDelete,
}) => {
  const [{ isDragging }, drag] = useDrag({
    type: 'grid-item',
    item: { index },
    collect: (monitor) => ({
      isDragging: monitor.isDragging(),
    }),
  });

  const [, drop] = useDrop({
    accept: 'grid-item',
    hover: (draggedItem: { index: number }) => {
      if (draggedItem.index !== index) {
        onMove(draggedItem.index, index);
        draggedItem.index = index;
      }
    },
  });

  return (
    <Col xs={24} sm={12} md={8} lg={6} xl={4}>
      <div ref={(node) => drag(drop(node))}>
        <Card
          size="small"
          hoverable
          className={`h-full transition-opacity ${isDragging ? 'opacity-50' : ''}`}
          extra={
            <Button
              type="text"
              size="small"
              icon={<HolderOutlined />}
              className="cursor-move"
            />
          }
          actions={[
            <Button
              key="edit"
              type="link"
              size="small"
              onClick={() => onEdit(item)}
            >
              编辑
            </Button>,
            <Button
              key="delete"
              type="link"
              size="small"
              danger
              onClick={() => onDelete(item.id)}
            >
              删除
            </Button>,
          ]}
        >
          <Card.Meta
            title={item.name}
            description={item.description}
          />
        </Card>
      </div>
    </Col>
  );
};

interface DraggableGridViewProps {
  data: {Entity}[];
  onReorder: (newData: {Entity}[]) => void;
  onEdit: (item: {Entity}) => void;
  onDelete: (id: string) => void;
}

export const DraggableGridView: React.FC<DraggableGridViewProps> = ({
  data,
  onReorder,
  onEdit,
  onDelete,
}) => {
  const [items, setItems] = useState(data);

  const moveItem = (dragIndex: number, hoverIndex: number) => {
    const newItems = [...items];
    const draggedItem = newItems[dragIndex];
    newItems.splice(dragIndex, 1);
    newItems.splice(hoverIndex, 0, draggedItem);
    setItems(newItems);
    onReorder(newItems);
  };

  return (
    <DndProvider backend={HTML5Backend}>
      <Row gutter={[16, 16]}>
        {items.map((item, index) => (
          <DraggableGridItem
            key={item.id}
            item={item}
            index={index}
            onMove={moveItem}
            onEdit={onEdit}
            onDelete={onDelete}
          />
        ))}
      </Row>
    </DndProvider>
  );
};
```
