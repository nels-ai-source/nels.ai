# Nels.AI 前端开发指引 - 左树右表格模式

## 1. 适用场景
- **权限管理**: 左侧菜单树，右侧权限表格
- **组织管理**: 左侧部门树，右侧员工表格
- **分类数据**: 左侧分类树，右侧数据表格
- **系统配置**: 左侧配置分组，右侧配置项表格

## 2. 页面特点
- **树形导航**: 左侧树形结构导航
- **表格展示**: 右侧表格形式展示详细数据
- **联动交互**: 左右两侧数据联动
- **批量操作**: 支持表格批量操作

## 3. 页面布局组件模板

### 3.1 基础左树右表格页面
```typescript:src/pages/{Module}/tree-table.tsx
import React, { useState, useEffect, useRef } from 'react';
import { PageContainer, ProTable } from '@ant-design/pro-components';
import { Row, Col, Card, Tree, Button, Space, Popconfirm, message } from 'antd';
import { PlusOutlined, EditOutlined, DeleteOutlined, ReloadOutlined } from '@ant-design/icons';
import { useIntl } from '@umijs/max';
import type { ActionType, ProColumns } from '@ant-design/pro-components';
import type { DataNode } from 'antd/es/tree';

import { use{Entity}Tree } from '@/hooks/use{Entity}Tree';
import { use{Entity}Table } from '@/hooks/use{Entity}Table';
import { TreeToolbar } from './components/tree-toolbar';
import { CreateModal } from './components/create-modal';
import type { {Entity}, {Entity}TreeNode } from '@/types/{module}';

const {Entity}TreeTable: React.FC = () => {
  const intl = useIntl();
  const actionRef = useRef<ActionType>();
  const [selectedTreeNode, setSelectedTreeNode] = useState<string | null>(null);
  const [createModalOpen, setCreateModalOpen] = useState(false);
  const [editingItem, setEditingItem] = useState<{Entity} | null>(null);
  const [selectedRowKeys, setSelectedRowKeys] = useState<React.Key[]>([]);

  const {
    treeData,
    loading: treeLoading,
    loadTree,
    addTreeNode,
    updateTreeNode,
    deleteTreeNode,
  } = use{Entity}Tree();

  const {
    loadTableData,
    deleteItem,
    batchDelete,
  } = use{Entity}Table();

  // 初始化加载树数据
  useEffect(() => {
    loadTree();
  }, [loadTree]);

  // 树节点选择
  const handleTreeSelect = (selectedKeys: React.Key[]) => {
    const nodeId = selectedKeys[0] as string;
    setSelectedTreeNode(nodeId);
    setSelectedRowKeys([]); // 清空表格选择
    if (actionRef.current) {
      actionRef.current.reload();
    }
  };

  // 树数据转换
  const convertToTreeData = (nodes: {Entity}TreeNode[]): DataNode[] => {
    return nodes.map(node => ({
      key: node.id,
      title: node.name,
      children: node.children ? convertToTreeData(node.children) : undefined,
      isLeaf: !node.children || node.children.length === 0,
    }));
  };

  // 表格列定义
  const columns: ProColumns<{Entity}>[] = [
    {
      title: intl.formatMessage({ id: 'fields.name' }),
      dataIndex: 'name',
      key: 'name',
      ellipsis: true,
      copyable: true,
      formItemProps: {
        rules: [
          {
            required: true,
            message: intl.formatMessage({ id: 'validation.required' }),
          },
        ],
      },
    },
    {
      title: intl.formatMessage({ id: 'fields.type' }),
      dataIndex: 'type',
      key: 'type',
      valueType: 'select',
      valueEnum: {
        1: { text: '类型1', status: 'Default' },
        2: { text: '类型2', status: 'Processing' },
        3: { text: '类型3', status: 'Success' },
      },
      filters: true,
      onFilter: true,
    },
    {
      title: intl.formatMessage({ id: 'fields.status' }),
      dataIndex: 'status',
      key: 'status',
      valueType: 'select',
      valueEnum: {
        1: { text: '活跃', status: 'Success' },
        2: { text: '非活跃', status: 'Default' },
        3: { text: '待审核', status: 'Processing' },
      },
      filters: true,
      onFilter: true,
    },
    {
      title: intl.formatMessage({ id: 'fields.description' }),
      dataIndex: 'description',
      key: 'description',
      ellipsis: true,
      search: false,
      width: 200,
    },
    {
      title: intl.formatMessage({ id: 'fields.creationTime' }),
      dataIndex: 'creationTime',
      key: 'creationTime',
      valueType: 'dateTime',
      sorter: true,
      search: false,
      width: 180,
    },
    {
      title: intl.formatMessage({ id: 'actions.title' }),
      key: 'actions',
      fixed: 'right',
      width: 150,
      search: false,
      render: (_, record) => (
        <Space>
          <Button
            type="link"
            size="small"
            icon={<EditOutlined />}
            onClick={() => setEditingItem(record)}
          >
            {intl.formatMessage({ id: 'actions.edit' })}
          </Button>
          <Popconfirm
            title={intl.formatMessage({ id: 'deleteConfirm.title' })}
            description={intl.formatMessage({ id: 'deleteConfirm.content' })}
            onConfirm={() => deleteItem(record.id)}
            okText={intl.formatMessage({ id: 'actions.confirm' })}
            cancelText={intl.formatMessage({ id: 'actions.cancel' })}
          >
            <Button
              type="link"
              size="small"
              danger
              icon={<DeleteOutlined />}
            >
              {intl.formatMessage({ id: 'actions.delete' })}
            </Button>
          </Popconfirm>
        </Space>
      ),
    },
  ];

  // 批量删除
  const handleBatchDelete = async () => {
    if (selectedRowKeys.length === 0) {
      message.warning('请选择要删除的项目');
      return;
    }

    try {
      await batchDelete(selectedRowKeys as string[]);
      setSelectedRowKeys([]);
      actionRef.current?.reload();
    } catch (error) {
      // 错误已在hook中处理
    }
  };

  return (
    <PageContainer>
      <Row gutter={16} style={{ height: 'calc(100vh - 200px)' }}>
        {/* 左侧树形导航 */}
        <Col span={6}>
          <Card
            title={intl.formatMessage({ id: '{module}.tree.title' })}
            size="small"
            className="h-full"
            extra={
              <TreeToolbar
                onAdd={() => setCreateModalOpen(true)}
                onRefresh={() => loadTree()}
                selectedNode={selectedTreeNode}
                onEdit={(node) => setEditingItem(node)}
                onDelete={(nodeId) => deleteTreeNode(nodeId)}
              />
            }
          >
            <div className="h-full overflow-auto">
              {treeData.length > 0 ? (
                <Tree
                  treeData={convertToTreeData(treeData)}
                  onSelect={handleTreeSelect}
                  selectedKeys={selectedTreeNode ? [selectedTreeNode] : []}
                  showLine
                  showIcon={false}
                  blockNode
                  loading={treeLoading}
                />
              ) : (
                <div className="flex items-center justify-center h-full text-gray-500">
                  {treeLoading ? '加载中...' : '暂无数据'}
                </div>
              )}
            </div>
          </Card>
        </Col>

        {/* 右侧表格内容 */}
        <Col span={18}>
          <ProTable<{Entity}>
            headerTitle={
              selectedTreeNode
                ? intl.formatMessage({ id: '{module}.table.title' })
                : intl.formatMessage({ id: '{module}.table.selectNode' })
            }
            actionRef={actionRef}
            rowKey="id"
            search={{
              labelWidth: 'auto',
              defaultCollapsed: false,
            }}
            toolBarRender={() => [
              <Button
                key="create"
                icon={<PlusOutlined />}
                type="primary"
                onClick={() => setCreateModalOpen(true)}
                disabled={!selectedTreeNode}
              >
                {intl.formatMessage({ id: 'actions.create' })}
              </Button>,
              <Button
                key="refresh"
                icon={<ReloadOutlined />}
                onClick={() => actionRef.current?.reload()}
              >
                {intl.formatMessage({ id: 'actions.refresh' })}
              </Button>,
            ]}
            request={async (params, sort, filter) => {
              if (!selectedTreeNode) {
                return {
                  data: [],
                  success: true,
                  total: 0,
                };
              }

              const queryParams = {
                ...params,
                categoryId: selectedTreeNode,
                skipCount: ((params.current || 1) - 1) * (params.pageSize || 20),
                maxResultCount: params.pageSize || 20,
              };
              
              return await loadTableData(queryParams);
            }}
            columns={columns}
            rowSelection={{
              selectedRowKeys,
              onChange: setSelectedRowKeys,
            }}
            tableAlertRender={({ selectedRowKeys, onCleanSelected }) => (
              <Space size={24}>
                <span>
                  已选择 <strong>{selectedRowKeys.length}</strong> 项
                  <a style={{ marginLeft: 8 }} onClick={onCleanSelected}>
                    取消选择
                  </a>
                </span>
              </Space>
            )}
            tableAlertOptionRender={({ selectedRowKeys }) => (
              <Space size={16}>
                <Button
                  size="small"
                  danger
                  onClick={handleBatchDelete}
                  disabled={selectedRowKeys.length === 0}
                >
                  批量删除 ({selectedRowKeys.length})
                </Button>
              </Space>
            )}
            pagination={{
              showQuickJumper: true,
              showSizeChanger: true,
              showTotal: (total, range) =>
                `第 ${range[0]}-${range[1]} 条/总共 ${total} 条`,
            }}
            scroll={{ x: 1000 }}
            size="small"
          />
        </Col>
      </Row>

      <CreateModal
        open={createModalOpen || !!editingItem}
        values={editingItem}
        type={editingItem ? 'edit' : 'create'}
        categoryId={selectedTreeNode}
        onOpenChange={(open) => {
          if (!open) {
            setCreateModalOpen(false);
            setEditingItem(null);
          }
        }}
        onSuccess={() => {
          actionRef.current?.reload();
          loadTree(); // 如果是树节点操作，需要刷新树
          setCreateModalOpen(false);
          setEditingItem(null);
        }}
      />
    </PageContainer>
  );
};

export default {Entity}TreeTable;
```

## 4. 专用Hooks

### 4.1 树表联动Hook
```typescript:src/hooks/use{Entity}TreeTable.ts
import { useState, useCallback, useEffect } from 'react';
import { message } from 'antd';
import { 
  get{Entity}Tree, 
  get{Entity}List, 
  delete{Entity}, 
  delete{Entity}Batch 
} from '@/services/{module}';
import type { {Entity}, {Entity}TreeNode, {Entity}Filter } from '@/types/{module}';

export function use{Entity}TreeTable() {
  const [treeData, setTreeData] = useState<{Entity}TreeNode[]>([]);
  const [treeLoading, setTreeLoading] = useState(false);
  const [selectedTreeNode, setSelectedTreeNode] = useState<string | null>(null);
  const [tableData, setTableData] = useState<{Entity}[]>([]);
  const [tableLoading, setTableLoading] = useState(false);
  const [total, setTotal] = useState(0);

  // 加载树数据
  const loadTree = useCallback(async () => {
    try {
      setTreeLoading(true);
      const result = await get{Entity}Tree();
      setTreeData(result);
    } catch (error) {
      message.error('加载树数据失败');
      console.error(error);
    } finally {
      setTreeLoading(false);
    }
  }, []);

  // 加载表格数据
  const loadTableData = useCallback(async (params: any) => {
    if (!selectedTreeNode) {
      return {
        data: [],
        success: true,
        total: 0,
      };
    }

    try {
      setTableLoading(true);
      const filter: {Entity}Filter = {
        ...params,
        categoryId: selectedTreeNode,
      };

      const result = await get{Entity}List(filter);
      setTableData(result.items);
      setTotal(result.totalCount);
      
      return {
        data: result.items,
        success: true,
        total: result.totalCount,
      };
    } catch (error) {
      message.error('加载表格数据失败');
      console.error(error);
      return {
        data: [],
        success: false,
        total: 0,
      };
    } finally {
      setTableLoading(false);
    }
  }, [selectedTreeNode]);

  // 树节点选择
  const handleTreeSelect = useCallback((nodeId: string | null) => {
    setSelectedTreeNode(nodeId);
  }, []);

  // 删除表格项目
  const deleteTableItem = useCallback(async (id: string) => {
    try {
      await delete{Entity}(id);
      message.success('删除成功');
      return true;
    } catch (error) {
      message.error('删除失败');
      console.error(error);
      return false;
    }
  }, []);

  // 批量删除表格项目
  const batchDeleteTableItems = useCallback(async (ids: string[]) => {
    try {
      await delete{Entity}Batch(ids);
      message.success(`成功删除 ${ids.length} 个项目`);
      return true;
    } catch (error) {
      message.error('批量删除失败');
      console.error(error);
      return false;
    }
  }, []);

  // 刷新当前数据
  const refresh = useCallback(async () => {
    await Promise.all([
      loadTree(),
      selectedTreeNode ? loadTableData({}) : Promise.resolve(),
    ]);
  }, [loadTree, loadTableData, selectedTreeNode]);

  return {
    // 树相关
    treeData,
    treeLoading,
    loadTree,
    
    // 表格相关
    tableData,
    tableLoading,
    total,
    loadTableData,
    deleteTableItem,
    batchDeleteTableItems,
    
    // 联动相关
    selectedTreeNode,
    handleTreeSelect,
    refresh,
  };
}
```

### 4.2 权限管理Hook
```typescript:src/hooks/use{Entity}Permission.ts
import { useState, useCallback } from 'react';
import { message } from 'antd';
import { 
  get{Entity}Permissions, 
  update{Entity}Permissions,
  batchUpdate{Entity}Permissions 
} from '@/services/{module}';
import type { {Entity}Permission } from '@/types/{module}';

export function use{Entity}Permission() {
  const [permissions, setPermissions] = useState<{Entity}Permission[]>([]);
  const [loading, setLoading] = useState(false);

  // 获取权限列表
  const loadPermissions = useCallback(async (entityId: string) => {
    try {
      setLoading(true);
      const result = await get{Entity}Permissions(entityId);
      setPermissions(result);
    } catch (error) {
      message.error('加载权限失败');
      console.error(error);
    } finally {
      setLoading(false);
    }
  }, []);

  // 更新权限
  const updatePermissions = useCallback(async (
    entityId: string, 
    permissions: string[]
  ) => {
    try {
      await update{Entity}Permissions(entityId, permissions);
      message.success('权限更新成功');
      return true;
    } catch (error) {
      message.error('权限更新失败');
      console.error(error);
      return false;
    }
  }, []);

  // 批量更新权限
  const batchUpdatePermissions = useCallback(async (
    entityIds: string[], 
    permissions: string[]
  ) => {
    try {
      await batchUpdate{Entity}Permissions(entityIds, permissions);
      message.success(`成功更新 ${entityIds.length} 个项目的权限`);
      return true;
    } catch (error) {
      message.error('批量权限更新失败');
      console.error(error);
      return false;
    }
  }, []);

  return {
    permissions,
    loading,
    loadPermissions,
    updatePermissions,
    batchUpdatePermissions,
  };
}

export const PermissionModal: React.FC<PermissionModalProps> = ({

}) => {
  const intl = useIntl();
  const [checkedKeys, setCheckedKeys] = useState<React.Key[]>([]);
  const [expandedKeys, setExpandedKeys] = useState<React.Key[]>([]);
  const [submitting, setSubmitting] = useState(false);
  
  const { permissions, loading, loadPermissions, updatePermissions, batchUpdatePermissions } = use{Entity}Permission();

  // 权限树数据
  const permissionTreeData: DataNode[] = [
    {
      title: '系统管理',
      key: 'system',
      children: [
        { title: '用户管理', key: 'system.user' },
        { title: '角色管理', key: 'system.role' },
        { title: '权限管理', key: 'system.permission' },
      ],
    },
    {
      title: '内容管理',
      key: 'content',
      children: [
        { title: '文章管理', key: 'content.article' },
        { title: '分类管理', key: 'content.category' },
        { title: '标签管理', key: 'content.tag' },
      ],
    },
    {
      title: '数据统计',
      key: 'analytics',
      children: [
        { title: '访问统计', key: 'analytics.visit' },
        { title: '用户统计', key: 'analytics.user' },
        { title: '内容统计', key: 'analytics.content' },
      ],
    },
  ];

  // 加载当前权限
  useEffect(() => {
    if (open && entityId) {
      loadPermissions(entityId);
    }
  }, [open, entityId, loadPermissions]);

  // 设置已选权限
  useEffect(() => {
    if (permissions.length > 0) {
      const keys = permissions.map(p => p.permissionKey);
      setCheckedKeys(keys);
      
      // 自动展开有权限的节点
      const expandKeys = new Set<string>();
      keys.forEach(key => {
        const parts = key.toString().split('.');
        if (parts.length > 1) {
          expandKeys.add(parts[0]);
        }
      });
      setExpandedKeys(Array.from(expandKeys));
    }
  }, [permissions]);

  // 提交权限更新
  const handleSubmit = async () => {
    try {
      setSubmitting(true);
      const permissionKeys = checkedKeys as string[];
      
      let success = false;
      if (entityIds && entityIds.length > 0) {
        // 批量更新
        success = await batchUpdatePermissions(entityIds, permissionKeys);
      } else if (entityId) {
        // 单个更新
        success = await updatePermissions(entityId, permissionKeys);
      }
      
      if (success) {
        onSuccess();
        onOpenChange(false);
      }
    } catch (error) {
      message.error('权限设置失败');
    } finally {
      setSubmitting(false);
    }
  };

  const title = entityIds && entityIds.length > 0 
    ? `批量权限设置 (${entityIds.length} 项)`
    : '权限设置';

  return (
    <Modal
      title={title}
      open={open}
      onCancel={() => onOpenChange(false)}
      width={600}
      footer={[
        <Button key="cancel" onClick={() => onOpenChange(false)}>
          {intl.formatMessage({ id: 'actions.cancel' })}
        </Button>,
        <Button
          key="submit"
          type="primary"
          loading={submitting}
          onClick={handleSubmit}
        >
          {intl.formatMessage({ id: 'actions.confirm' })}
        </Button>,
      ]}
    >
      <Spin spinning={loading}>
        <div className="mb-4">
          <Space>
            <Button
              size="small"
              onClick={() => setExpandedKeys(permissionTreeData.map(item => item.key))}
            >
              展开全部
            </Button>
            <Button
              size="small"
              onClick={() => setExpandedKeys([])}
            >
              收起全部
            </Button>
            <Button
              size="small"
              onClick={() => setCheckedKeys([])}
            >
              清空选择
            </Button>
          </Space>
        </div>
        
        <Tree
          checkable
          treeData={permissionTreeData}
          checkedKeys={checkedKeys}
          expandedKeys={expandedKeys}
          onCheck={setCheckedKeys}
          onExpand={setExpandedKeys}
          height={400}
        />
        
        <div className="mt-4 text-gray-500 text-sm">
          已选择 {checkedKeys.length} 个权限
        </div>
      </Spin>
    </Modal>
  );
};
```
