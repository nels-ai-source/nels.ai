# Nels.AI 前端开发指引 - 数据流与状态管理

## 1. 状态管理架构

### 1.1 状态分层
```
全局状态 (Global State)
├── 用户状态 (User State)
├── 应用配置 (App Config)
├── 路由状态 (Router State)
└── 主题状态 (Theme State)

页面状态 (Page State)
├── 列表数据 (List Data)
├── 筛选条件 (Filter State)
├── 分页信息 (Pagination)
└── 选择状态 (Selection)

组件状态 (Component State)
├── 表单状态 (Form State)
├── UI状态 (UI State)
└── 临时状态 (Temp State)
```

### 1.2 状态管理工具选择
- **全局状态**: Zustand / Redux Toolkit
- **服务端状态**: TanStack Query (React Query)
- **表单状态**: Ant Design Form / React Hook Form
- **URL状态**: React Router + URLSearchParams

import { create } from 'zustand';
import { devtools, persist } from 'zustand/middleware';
import { immer } from 'zustand/middleware/immer';

interface User {
  id: string;
  name: string;
  email: string;
  avatar?: string;
  permissions: string[];
}

interface AppConfig {
  theme: 'light' | 'dark';
  language: 'zh-CN' | 'en-US';
  sidebarCollapsed: boolean;
}

interface AppState {
  // 用户状态
  user: User | null;
  isAuthenticated: boolean;
  
  // 应用配置
  config: AppConfig;
  
  // 加载状态
  loading: {
    user: boolean;
    app: boolean;
  };
  
  // Actions
  setUser: (user: User | null) => void;
  updateConfig: (config: Partial<AppConfig>) => void;
  setLoading: (key: keyof AppState['loading'], loading: boolean) => void;
  logout: () => void;
}

export const useAppStore = create<AppState>()(
  devtools(
    persist(
      immer((set, get) => ({
        // 初始状态
        user: null,
        isAuthenticated: false,
        config: {
          theme: 'light',
          language: 'zh-CN',
          sidebarCollapsed: false,
        },
        loading: {
          user: false,
          app: false,
        },

        // Actions
        setUser: (user) =>
          set((state) => {
            state.user = user;
            state.isAuthenticated = !!user;
          }),

        updateConfig: (newConfig) =>
          set((state) => {
            Object.assign(state.config, newConfig);
          }),

        setLoading: (key, loading) =>
          set((state) => {
            state.loading[key] = loading;
          }),

        logout: () =>
          set((state) => {
            state.user = null;
            state.isAuthenticated = false;
          }),
      })),
      {
        name: 'app-store',
        partialize: (state) => ({
          config: state.config,
          user: state.user,
          isAuthenticated: state.isAuthenticated,
        }),
      }
    ),
    { name: 'AppStore' }
  )
);

// 选择器
export const useUser = () => useAppStore((state) => state.user);
export const useIsAuthenticated = () => useAppStore((state) => state.isAuthenticated);
export const useAppConfig = () => useAppStore((state) => state.config);
export const useAppLoading = () => useAppStore((state) => state.loading);

### 2.2 实体状态管理模板
```typescript:src/stores/useEntityStore.ts
import { create } from 'zustand';
import { devtools } from 'zustand/middleware';
import { immer } from 'zustand/middleware/immer';

interface Entity {
  id: string;
  name: string;
  status: number;
  createdAt: string;
  updatedAt: string;
}

interface EntityFilter {
  name?: string;
  status?: number;
  dateRange?: [string, string];
}

interface EntityState {
  // 数据
  entities: Entity[];
  currentEntity: Entity | null;
  
  // 列表状态
  filter: EntityFilter;
  pagination: {
    current: number;
    pageSize: number;
    total: number;
  };
  
  // UI状态
  loading: {
    list: boolean;
    detail: boolean;
    create: boolean;
    update: boolean;
    delete: boolean;
  };
  
  // 选择状态
  selectedKeys: React.Key[];
  selectedRows: Entity[];
  
  // Actions
  setEntities: (entities: Entity[]) => void;
  setCurrentEntity: (entity: Entity | null) => void;
  updateEntity: (id: string, updates: Partial<Entity>) => void;
  removeEntity: (id: string) => void;
  setFilter: (filter: Partial<EntityFilter>) => void;
  setPagination: (pagination: Partial<EntityState['pagination']>) => void;
  setLoading: (key: keyof EntityState['loading'], loading: boolean) => void;
  setSelection: (keys: React.Key[], rows: Entity[]) => void;
  clearSelection: () => void;
  reset: () => void;
}

const initialState = {
  entities: [],
  currentEntity: null,
  filter: {},
  pagination: {
    current: 1,
    pageSize: 20,
    total: 0,
  },
  loading: {
    list: false,
    detail: false,
    create: false,
    update: false,
    delete: false,
  },
  selectedKeys: [],
  selectedRows: [],
};

export const useEntityStore = create<EntityState>()(
  devtools(
    immer((set, get) => ({
      ...initialState,

      setEntities: (entities) =>
        set((state) => {
          state.entities = entities;
        }),

      setCurrentEntity: (entity) =>
        set((state) => {
          state.currentEntity = entity;
        }),

      updateEntity: (id, updates) =>
        set((state) => {
          const index = state.entities.findIndex((item) => item.id === id);
          if (index !== -1) {
            Object.assign(state.entities[index], updates);
          }
          if (state.currentEntity?.id === id) {
            Object.assign(state.currentEntity, updates);
          }
        }),

      removeEntity: (id) =>
        set((state) => {
          state.entities = state.entities.filter((item) => item.id !== id);
          if (state.currentEntity?.id === id) {
            state.currentEntity = null;
          }
          state.selectedKeys = state.selectedKeys.filter((key) => key !== id);
          state.selectedRows = state.selectedRows.filter((row) => row.id !== id);
        }),

      setFilter: (filter) =>
        set((state) => {
          Object.assign(state.filter, filter);
          state.pagination.current = 1; // 重置页码
        }),

      setPagination: (pagination) =>
        set((state) => {
          Object.assign(state.pagination, pagination);
        }),

      setLoading: (key, loading) =>
        set((state) => {
          state.loading[key] = loading;
        }),

      setSelection: (keys, rows) =>
        set((state) => {
          state.selectedKeys = keys;
          state.selectedRows = rows;
        }),

      clearSelection: () =>
        set((state) => {
          state.selectedKeys = [];
          state.selectedRows = [];
        }),

      reset: () =>
        set((state) => {
          Object.assign(state, initialState);
        }),
    })),
    { name: 'EntityStore' }
  )
);

// 选择器
export const useEntities = () => useEntityStore((state) => state.entities);
export const useCurrentEntity = () => useEntityStore((state) => state.currentEntity);
export const useEntityFilter = () => useEntityStore((state) => state.filter);
export const useEntityPagination = () => useEntityStore((state) => state.pagination);
export const useEntityLoading = () => useEntityStore((state) => state.loading);
export const useEntitySelection = () => useEntityStore((state) => ({
  selectedKeys: state.selectedKeys,
  selectedRows: state.selectedRows,
}));
```

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { message } from 'antd';
import * as entityService from '@/services/entity';
import { useEntityStore } from '@/stores/useEntityStore';

// 查询键工厂
export const entityKeys = {
  all: ['entities'] as const,
  lists: () => [...entityKeys.all, 'list'] as const,
  list: (filter: any) => [...entityKeys.lists(), filter] as const,
  details: () => [...entityKeys.all, 'detail'] as const,
  detail: (id: string) => [...entityKeys.details(), id] as const,
};

// 列表查询
export const useEntityList = (filter: any, pagination: any) => {
  const setLoading = useEntityStore((state) => state.setLoading);
  
  return useQuery({
    queryKey: entityKeys.list({ filter, pagination }),
    queryFn: () => entityService.getEntityList({ ...filter, ...pagination }),
    onSuccess: (data) => {
      setLoading('list', false);
    },
    onError: () => {
      setLoading('list', false);
    },
  });
};

// 详情查询
export const useEntityDetail = (id: string, enabled = true) => {
  const setLoading = useEntityStore((state) => state.setLoading);
  
  return useQuery({
    queryKey: entityKeys.detail(id),
    queryFn: () => entityService.getEntity(id),
    enabled: enabled && !!id,
    onSuccess: () => {
      setLoading('detail', false);
    },
    onError: () => {
      setLoading('detail', false);
    },
  });
};

// 创建变更
export const useCreateEntity = () => {
  const queryClient = useQueryClient();
  const setLoading = useEntityStore((state) => state.setLoading);
  
  return useMutation({
    mutationFn: entityService.createEntity,
    onMutate: () => {
      setLoading('create', true);
    },
    onSuccess: () => {
      message.success('创建成功');
      queryClient.invalidateQueries({ queryKey: entityKeys.lists() });
      setLoading('create', false);
    },
    onError: () => {
      setLoading('create', false);
    },
  });
};

// 更新变更
export const useUpdateEntity = () => {
  const queryClient = useQueryClient();
  const setLoading = useEntityStore((state) => state.setLoading);
  const updateEntity = useEntityStore((state) => state.updateEntity);
  
  return useMutation({
    mutationFn: ({ id, data }: { id: string; data: any }) =>
      entityService.updateEntity(id, data),
    onMutate: () => {
      setLoading('update', true);
    },
    onSuccess: (data, variables) => {
      message.success('更新成功');
      // 乐观更新
      updateEntity(variables.id, data);
      // 使相关查询失效
      queryClient.invalidateQueries({ queryKey: entityKeys.detail(variables.id) });
      queryClient.invalidateQueries({ queryKey: entityKeys.lists() });
      setLoading('update', false);
    },
    onError: () => {
      setLoading('update', false);
    },
  });
};

// 删除变更
export const useDeleteEntity = () => {
  const queryClient = useQueryClient();
  const setLoading = useEntityStore((state) => state.setLoading);
  const removeEntity = useEntityStore((state) => state.removeEntity);
  
  return useMutation({
    mutationFn: entityService.deleteEntity,
    onMutate: () => {
      setLoading('delete', true);
    },
    onSuccess: (_, id) => {
      message.success('删除成功');
      // 乐观更新
      removeEntity(id);
      // 使相关查询失效
      queryClient.invalidateQueries({ queryKey: entityKeys.lists() });
      setLoading('delete', false);
    },
    onError: () => {
      setLoading('delete', false);
    },
  });
};

// 批量删除变更
export const useBatchDeleteEntity = () => {
  const queryClient = useQueryClient();
  const setLoading = useEntityStore((state) => state.setLoading);
  const clearSelection = useEntityStore((state) => state.clearSelection);
  
  return useMutation({
    mutationFn: entityService.batchDeleteEntity,
    onMutate: () => {
      setLoading('delete', true);
    },
    onSuccess: () => {
      message.success('批量删除成功');
      clearSelection();
      queryClient.invalidateQueries({ queryKey: entityKeys.lists() });
      setLoading('delete', false);
    },
    onError: () => {
      setLoading('delete', false);
    },
  });
};

## 4. 表单状态管理

### 4.1 表单 Hook 模板
```typescript:src/hooks/useEntityForm.ts
import { useState, useEffect } from 'react';
import { Form, message } from 'antd';
import { useCreateEntity, useUpdateEntity } from './useEntityQueries';

interface UseEntityFormProps {
  initialValues?: any;
  onSuccess?: () => void;
  mode?: 'create' | 'edit';
}

export const useEntityForm = ({
  initialValues,
  onSuccess,
  mode = 'create',
}: UseEntityFormProps = {}) => {
  const [form] = Form.useForm();
  const [submitting, setSubmitting] = useState(false);
  
  const createMutation = useCreateEntity();
  const updateMutation = useUpdateEntity();
  
  // 设置初始值
  useEffect(() => {
    if (initialValues) {
      form.setFieldsValue(initialValues);
    }
  }, [initialValues, form]);
  
  // 提交表单
  const handleSubmit = async (values?: any) => {
    try {
      const formValues = values || (await form.validateFields());
      setSubmitting(true);
      
      if (mode === 'create') {
        await createMutation.mutateAsync(formValues);
      } else {
        await updateMutation.mutateAsync({
          id: initialValues?.id,
          data: formValues,
        });
      }
      
      form.resetFields();
      onSuccess?.();
    } catch (error) {
      if (error.errorFields) {
        // 表单验证错误
        return;
      }
      console.error('Form submission error:', error);
    } finally {
      setSubmitting(false);
    }
  };
  
  // 重置表单
  const handleReset = () => {
    form.resetFields();
    if (initialValues) {
      form.setFieldsValue(initialValues);
    }
  };
  
  return {
    form,
    submitting: submitting || createMutation.isPending || updateMutation.isPending,
    handleSubmit,
    handleReset,
  };
};
```

### 4.2 搜索表单 Hook
```typescript:src/hooks/useSearchForm.ts
import { useState, useCallback, useEffect } from 'react';
import { Form } from 'antd';
import { useDebouncedCallback } from 'use-debounce';
import { useEntityStore } from '@/stores/useEntityStore';

interface UseSearchFormProps {
  defaultValues?: any;
  debounceMs?: number;
  autoSubmit?: boolean;
}

export const useSearchForm = ({
  defaultValues = {},
  debounceMs = 300,
  autoSubmit = false,
}: UseSearchFormProps = {}) => {
  const [form] = Form.useForm();
  const setFilter = useEntityStore((state) => state.setFilter);
  const filter = useEntityStore((state) => state.filter);
  
  // 防抖搜索
  const debouncedSearch = useDebouncedCallback(
    (values: any) => {
      setFilter(values);
    },
    debounceMs
  );
  
  // 处理搜索
  const handleSearch = useCallback((values?: any) => {
    const formValues = values || form.getFieldsValue();
    
    // 清理空值
    const cleanValues = Object.keys(formValues).reduce((acc, key) => {
      const value = formValues[key];
      if (value !== undefined && value !== null && value !== '') {
        acc[key] = value;
      }
      return acc;
    }, {} as any);
    
    if (autoSubmit) {
      debouncedSearch(cleanValues);
    } else {
      setFilter(cleanValues);
    }
  }, [form, setFilter, debouncedSearch, autoSubmit]);
  
  // 处理重置
  const handleReset = useCallback(() => {
    form.resetFields();
    setFilter({});
  }, [form, setFilter]);
  
  // 表单值变化时自动搜索
  const handleValuesChange = useCallback((changedValues: any, allValues: any) => {
    if (autoSubmit) {
      handleSearch(allValues);
    }
  }, [handleSearch, autoSubmit]);
  
  // 初始化表单值
  useEffect(() => {
    const initialValues = { ...defaultValues, ...filter };
    form.setFieldsValue(initialValues);
  }, [form, defaultValues, filter]);
  
  return {
    form,
    handleSearch,
    handleReset,
    handleValuesChange,
  };
};
```

## 5. URL状态同步

### 5.1 URL状态 Hook
```typescript:src/hooks/useUrlState.ts
import { useState, useEffect, useCallback } from 'react';
import { useSearchParams } from 'react-router-dom';

interface UseUrlStateOptions<T> {
  defaultValue?: T;
  serialize?: (value: T) => string;
  deserialize?: (value: string) => T;
}

export const useUrlState = <T>(
  key: string,
  options: UseUrlStateOptions<T> = {}
) => {
  const [searchParams, setSearchParams] = useSearchParams();
  const {
    defaultValue,
    serialize = JSON.stringify,
    deserialize = JSON.parse,
  } = options;
  
  const [state, setState] = useState<T>(() => {
    const urlValue = searchParams.get(key);
    if (urlValue) {
      try {
        return deserialize(urlValue);
      } catch {
        return defaultValue as T;
      }
    }
    return defaultValue as T;
  });
  
  const updateState = useCallback((value: T | ((prev: T) => T)) => {
    setState((prev) => {
      const newValue = typeof value === 'function' ? (value as (prev: T) => T)(prev) : value;
      
      // 更新URL
      const newSearchParams = new URLSearchParams(searchParams);
      if (newValue === undefined || newValue === null || newValue === defaultValue) {
        newSearchParams.delete(key);
      } else {
        try {
          newSearchParams.set(key, serialize(newValue));
        } catch {
          newSearchParams.delete(key);
        }
      }
      setSearchParams(newSearchParams, { replace: true });
      
      return newValue;
    });
  }, [key, searchParams, setSearchParams, serialize, defaultValue]);
  
  // 监听URL变化
  useEffect(() => {
    const urlValue = searchParams.get(key);
    if (urlValue) {
      try {
        const parsedValue = deserialize(urlValue);
        setState(parsedValue);
      } catch {
        setState(defaultValue as T);
      }
    } else {
      setState(defaultValue as T);
    }
  }, [searchParams, key, deserialize, defaultValue]);
  
  return [state, updateState] as const;
};
```

### 5.2 列表页面URL状态同步
```typescript:src/hooks/useListUrlState.ts
import { useEffect } from 'react';
import { useUrlState } from './useUrlState';
import { useEntityStore } from '@/stores/useEntityStore';

export const useListUrlState = () => {
  const filter = useEntityStore((state) => state.filter);
  const pagination = useEntityStore((state) => state.pagination);
  const setFilter = useEntityStore((state) => state.setFilter);
  const setPagination = useEntityStore((state) => state.setPagination);
  
  // URL状态
  const [urlFilter, setUrlFilter] = useUrlState('filter', {
    defaultValue: {},
  });
  
  const [urlPage, setUrlPage] = useUrlState('page', {
    defaultValue: 1,
    serialize: String,
    deserialize: Number,
  });
  
  const [urlPageSize, setUrlPageSize] = useUrlState('pageSize', {
    defaultValue: 20,
    serialize: String,
    deserialize: Number,
  });
  
  // 同步Store到URL
  useEffect(() => {
    setUrlFilter(filter);
  }, [filter, setUrlFilter]);
  
  useEffect(() => {
    setUrlPage(pagination.current);
  }, [pagination.current, setUrlPage]);
  
  useEffect(() => {
    setUrlPageSize(pagination.pageSize);
  }, [pagination.pageSize, setUrlPageSize]);
  
  // 同步URL到Store
  useEffect(() => {
    setFilter(urlFilter);
  }, [urlFilter, setFilter]);
  
  useEffect(() => {
    setPagination({
      current: urlPage,
      pageSize: urlPageSize,
    });
  }, [urlPage, urlPageSize, setPagination]);
};
```

## 6. 数据流最佳实践

### 6.1 数据流向
```
用户操作 → 组件事件 → Store Action → API调用 → 数据更新 → UI重新渲染
```

### 6.2 状态更新策略
1. **乐观更新**: 立即更新UI，失败时回滚
2. **悲观更新**: 等待API响应后更新UI
3. **缓存失效**: 使相关查询缓存失效
4. **局部更新**: 只更新变化的部分

### 6.3 错误处理
```typescript:src/utils/errorHandler.ts
import { message } from 'antd';

export const handleApiError = (error: any) => {
  if (error.response) {
    const { status, data } = error.response;
    
    switch (status) {
      case 401:
        message.error('登录已过期，请重新登录');
        // 跳转到登录页
        break;
      case 403:
        message.error('没有权限执行此操作');
        break;
      case 404:
        message.error('请求的资源不存在');
        break;
      case 500:
        message.error('服务器内部错误');
        break;
      default:
        message.error(data?.message || '操作失败');
    }
  } else {
    message.error('网络错误，请检查网络连接');
  }
};
```

### 6.4 性能优化
1. **选择器优化**: 使用精确的选择器避免不必要的重渲染
2. **缓存策略**: 合理设置查询缓存时间
3. **分页加载**: 大数据集使用分页或虚拟滚动
4. **防抖节流**: 搜索和筛选使用防抖

这个数据流与状态管理指引提供了完整的状态管理解决方案，确保数据流的清晰和可维护性。
